def greet(name, emoji="👋"): """دالة للترحيب بالأشخاص""" message = f"{emoji} مرحباً يا {name}!" return message print(greet("سارة")) print(greet("أحمد", "🌟")) def calculate_bmi(weight, height): """حساب مؤشر كتلة الجسم""" bmi = weight / (height ** 2) if bmi < 18.5: status = "نحيف" elif bmi < 25: status = "طبيعي ✅" else: status = "زيادة في الوزن" return bmi, status bmi, s = calculate_bmi(60, 1.70) print(f"BMI: {bmi:.1f} — {s}")

from sklearn.neighbors import KNeighborsClassifier X_train = [[1.8, 80], [1.6, 60], [1.75, 75], [1.55, 50]] y_train = ["ذكر", "أنثى", "ذكر", "أنثى"] model = KNeighborsClassifier(n_neighbors=3) model.fit(X_train, y_train) new_person = [[1.70, 70]] prediction = model.predict(new_person) print(f"التنبؤ: {prediction[0]}")

emotion_detect.py
from deepface import DeepFace import cv2 cap = cv2.VideoCapture(0) while True: ret, frame = cap.read() try: result = DeepFace.analyze( frame, actions=['emotion'], enforce_detection=False ) emotion = result[0]['dominant_emotion'] confidence = result[0]['emotion'][emotion] cv2.putText(frame, f"{emotion}: {confidence:.1f}%", (30, 50), cv2.FONT_HERSHEY_SIMPLEX, 1, (0, 255, 0), 2) except: pass cv2.imshow('Emotion Detection', frame) if cv2.waitKey(1) == ord('q'): break
