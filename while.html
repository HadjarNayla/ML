if / elif / else
if
الشرط الأساسي. إذا كان صحيحاً، نفّذ الكود.
elif
شرط إضافي يُختبر إذا فشل الأول.

if score >= 90:   print("ممتاز 🌟")

else:            print("راجعي الدرس 📚")

القائمة — List

القائمة هي حاوية مرتّبة قابلة للتعديل — يمكن إضافة عناصر، حذفها، وتغييرها في أي وقت. تُكتب بأقواس مربّعة [ ].

list_basics.py
fruits = ["تفاح", "موز", "برتقال", "عنب"]

# الوصول بالفهرس (يبدأ من 0)
print(fruits[0])    # تفاح
print(fruits[-1])   # عنب ← آخر عنصر

# التعديل
fruits[1] = "مانجو"   # تغيير قيمة
fruits.append("كيوي")  # إضافة للنهاية
fruits.remove("تفاح") # حذف بالقيمة

# التقطيع
print(fruits[1:3])   # ['موز','برتقال']
print(fruits[::-1])  # عكس القائمة

# دوال مدمجة
scores = [85, 92, 78, 96]
print(sum(scores), max(scores), min(scores))
fruits = ["تفاح","موز","برتقال","عنب"]

الصف — Tuple

الصف هو مثل القائمة تماماً لكنه ثابت لا يتغير بعد الإنشاء. يُكتب بأقواس عادية ( ). يُستخدم لتخزين بيانات لا يجب تغييرها.

tuple_basics.py
point   = (10, 20)            # إحداثيات
colors  = ("أحمر", "أخضر", "أزرق")
single  = (42,)              # صف بعنصر واحد (الفاصلة ضرورية)

# الوصول مثل القائمة
print(point[0])    # 10
print(colors[-1]) # أزرق

# التفكيك (Unpacking)
x, y = point
print(x, y)   # 10  20

# ثابت — هذا سيُطلق TypeError!
# point[0] = 99   ← خطأ!

# الدالة ترجع صفاً
def min_max(lst):
    return min(lst), max(lst)

lo, hi = min_max([3,1,8,2])   # 1  8
colors = ("أحمر", "أخضر", "أزرق")

dict_basics.py
student = {

print(student["name"])              # Naïla

for key, val in student.items():
    print(f"{key}: {val}")

المجموعة — Set

المجموعة هي تجميعة عناصر فريدة غير مرتّبة — تُزيل التكرار تلقائياً. تُكتب بأقواس معقوفة { } لكن بدون مفاتيح. تدعم العمليات الرياضية كالاتحاد والتقاطع.

set_basics.py
# إنشاء مجموعة
nums = {1, 2, 3, 2, 1}
print(nums)   # {1, 2, 3} ← التكرار أُزيل

# من قائمة (إزالة المكررات)
lst = [1, 2, 2, 3, 3, 3]
unique = set(lst)   # {1, 2, 3}

# إضافة وحذف
nums.add(4)
nums.remove(2)
print(3 in nums)   # True

# العمليات الرياضية
A = {1, 2, 3, 4}
B = {3, 4, 5, 6}

print(A | B)   # الاتحاد:   {1,2,3,4,5,6}
print(A & B)   # التقاطع:  {3,4}
print(A - B)   # الفرق:    {1,2}
print(A ^ B)   # التماثل:  {1,2,5,6}
